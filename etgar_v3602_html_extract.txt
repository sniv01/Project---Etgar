  function rotr(x, n) { return (x >>> n) | (x << (32 - n)); }

  function utf8Encode(str) {
    if (typeof TextEncoder !== 'undefined') return new TextEncoder().encode(str);
    var bytes = [];
    for (var i = 0; i < str.length; i++) {
      var cp = str.charCodeAt(i);
      if (cp >= 0xd800 && cp <= 0xdbff && i + 1 < str.length) {
        var lo = str.charCodeAt(++i);
        cp = ((cp - 0xd800) << 10) + (lo - 0xdc00) + 0x10000;
      }
      if (cp < 0x80) bytes.push(cp);
      else if (cp < 0x800) bytes.push(0xc0 | (cp >> 6), 0x80 | (cp & 0x3f));
      else if (cp < 0x10000) bytes.push(0xe0 | (cp >> 12), 0x80 | ((cp >> 6) & 0x3f), 0x80 | (cp & 0x3f));
      else bytes.push(0xf0 | (cp >> 18), 0x80 | ((cp >> 12) & 0x3f), 0x80 | ((cp >> 6) & 0x3f), 0x80 | (cp & 0x3f));

  function sha256Bytes(bytes) {
    var src = bytes instanceof Uint8Array ? bytes : new Uint8Array(bytes);
    var len = src.length;
    var padLen = ((len + 9 + 63) >> 6) << 6;
    var buf = new Uint8Array(padLen);
    buf.set(src);
    buf[len] = 0x80;
    var dv = new DataView(buf.buffer);
    dv.setUint32(padLen - 8, Math.floor(len / 0x20000000), false);
    dv.setUint32(padLen - 4, (len << 3) >>> 0, false);

    var h0 = 0x6a09e667, h1 = 0xbb67ae85, h2 = 0x3c6ef372, h3 = 0xa54ff53a;
    var h4 = 0x510e527f, h5 = 0x9b05688c, h6 = 0x1f83d9ab, h7 = 0x5be0cd19;

        var s1 = rotr(w2, 17) ^ rotr(w2, 19) ^ (w2 >>> 10);
        W[i] = (W[i - 16] + s0 + W[i - 7] + s1) >>> 0;
      }
      var a = h0, b = h1, c = h2, d = h3, e = h4, f = h5, g = h6, h = h7;
      for (var i = 0; i < 64; i++) {

        var ch = (e & f) ^ ((~e) & g);
        var temp1 = (h + S1 + ch + K[i] + W[i]) >>> 0;
        var S0 = rotr(a, 2) ^ rotr(a, 13) ^ rotr(a, 22);
        var maj = (a & b) ^ (a & c) ^ (b & c);
        var temp2 = (S0 + maj) >>> 0;
        h = g; g = f; f = e; e = (d + temp1) >>> 0;
        d = c; c = b; b = a; a = (temp1 + temp2) >>> 0;
      }
      h0 = (h0 + a) >>> 0; h1 = (h1 + b) >>> 0; h2 = (h2 + c) >>> 0; h3 = (h3 + d) >>> 0;
      h4 = (h4 + e) >>> 0; h5 = (h5 + f) >>> 0; h6 = (h6 + g) >>> 0; h7 = (h7 + h) >>> 0;
    }

    return [h0, h1, h2, h3, h4, h5, h6, h7].map(function (v) { return v.toString(16).padStart(8, '0'); }).join('');

  function canonicalJsonString(val) {
    if (val === null) return 'null';
    if (typeof val === 'number') {
      if (!isFinite(val)) throw new Error('Non-finite numbers are forbidden in RFC 8785 JCS.');
      return Object.is(val, -0) ? '0' : String(val);
    }
    if (typeof val !== 'object') return JSON.stringify(val);
    if (Array.isArray(val)) return '[' + val.map(canonicalJsonString).join(',') + ']';
    var keys = Object.keys(val).sort();
    var parts = [];
    for (var i = 0; i < keys.length; i++) {
      var k = keys[i];
      if (val[k] === undefined) continue;
      parts.push(JSON.stringify(k) + ':' + canonicalJsonString(val[k]));
    }
    return '{' + parts.join(',') + '}';
  }

  function handleFileImport(inputEl) {
    var file = inputEl && inputEl.files && inputEl.files[0];
    if (!file) return;
    var reader = new FileReader();
    reader.onload = function () {
      var rawBytes = new Uint8Array(reader.result);
      lastImportedFileBytes = rawBytes;
      lastImportedFileName = file.name;
      var ver = verifyRawBriefingBytes(rawBytes, null);
      if (!ver.ok) {
        showInboxStatus('❌ הייבוא נחסם [' + ver.code + ']: ' + ver.error, true);
      } else {
        var res = importBriefingObject(ver.doc, rawBytes, file.name, null);
        if (!res.ok) {
          showInboxStatus('❌ הייבוא נחסם [' + res.code + ']: ' + res.error, true);
        } else {
          showInboxStatus('✅ התדריך יובא לתיבת הדואר הנכנס (RFC 8785 SHA-256 מאומת · ' + res.import_id + ' · אירוע ' + res.import_event_id + '): נוספו +' + res.added + ', RETRACTED +' + res.retracted_added + ', דולגו כפילויות ' + res.skipped_duplicate + '. ניתן כעת לאמת גם קובץ .sha256.', false);
        }
      }
      inputEl.value = '';
    };
    reader.readAsArrayBuffer(file);
  }

  function handleSidecarFile(inputEl) {
    var file = inputEl && inputEl.files && inputEl.files[0];
    if (!file) return;
    if (!lastImportedFileBytes) {
      showInboxStatus('⚠️ יש לייבא תחילה את קובץ ה-JSON, ורק אז לטעון את קובץ ה-sidecar (.sha256) שלו לאימות הבתים הגולמיים.', true);
      inputEl.value = '';
      return;
    }
    var reader = new FileReader();
    reader.onload = function () {
      var sidecarTxt = String(reader.result || '');
      var ver = verifyRawBriefingBytes(lastImportedFileBytes, sidecarTxt);
      if (!ver.ok) {
        showInboxStatus('❌ אימות Sidecar נכשל [' + ver.code + ']: ' + ver.error, true);
      } else {
        showInboxStatus('✅ אימות Sidecar (.sha256) הצליח במלואו על הקובץ "' + lastImportedFileName + '"! File SHA-256: ' + ver.rawFileSha256.slice(0, 16) + '… | Payload SHA-256: ' + ver.payloadSha256.slice(0, 16) + '…', false);
      }
      inputEl.value = '';
    };
    reader.readAsText(file, 'utf-8');
  }

window.v360HumanTransferFromInbox = function (idx, ev, isRetracted, skipConfirmForTest) {
  if (ev && ev.isTrusted === false) return false;
  var item = window.v360EtgarInbox && window.v360EtgarInbox.getItemSnapshot(idx, !!isRetracted);
  if (!item) return false;
  if (!skipConfirmForTest && typeof window !== 'undefined' && typeof window.confirm === 'function') {
    var ok = window.confirm(
      'אישור העברה אנושית מתיבת אתגר (EtgarInbox) אל טאב 03 (מכונת רנטגן):\n\n' +
      '• פריט: ' + (item.title || '') + '\n' +
      '• מזהה: ' + (item.item_id || '') + ' | סטטוס משיכה: ' + (item.retraction_status || 'UNKNOWN') + '\n' +
      '• יעד: טאב 03 (מכונת רנטגן בלבד — אינו כותב לארכיון או לכספת ההפרכות)\n\n' +
      'להמשיך בהעברה?'
    );
    if (!ok) return false;
  }
  var titleEl = document.getElementById('xrayTitle');
  var inputEl = document.getElementById('xrayInput');
  var urlEl = document.getElementById('xrayUrlInput');
  if (titleEl) titleEl.value = (isRetracted ? '[RETRACTED_SOURCE] ' : '') + (item.title || '');
  if (inputEl) inputEl.value = item.raw_text || '';
  if (urlEl) urlEl.value = (item.provenance && item.provenance.original_url) || '';
  if (typeof saveDraftState === 'function') saveDraftState();
  if (typeof switchTab === 'function') switchTab('xray');
  if (typeof runLocalXray === 'function') runLocalXray();
  return true;
};
